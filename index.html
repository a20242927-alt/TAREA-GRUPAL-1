import mapclassify
import numpy as np

np.random.seed(12345) # Fijamos la semilla para obtener siempre los mismos resultados

# Definimos 5 intervalos para la clasificación
K = 5

# Eliminamos valores nulos de la serie para asegurar el cálculo del ADCM
theVar = mapaProv['empleo_poblacion'].dropna()

# Mismo ancho de intervalo, interpretación sencilla
ei5 = mapclassify.EqualInterval(theVar, k=K)

# Intervalos basados en desviación estándar y media
msd = mapclassify.StdMean(theVar)

# Cuantiles
q5 = mapclassify.Quantiles(theVar, k=K)

# Basado en similitud, ideal para datos multimodales
mb5 = mapclassify.MaximumBreaks(theVar, k=K)

# Basado en similitud, excelente para datos sesgados
ht = mapclassify.HeadTailBreaks(theVar)

# Algoritmos de optimización
fj5 = mapclassify.FisherJenks(theVar, k=K)
jc5 = mapclassify.JenksCaspall(theVar, k=K)

Para escoger la mejor clasificación usamos las desviación absoluta y la comparamos con la mediana.
